import { AvatarDemo } from "@/components/ui/avatar/avatar-demo";

The `Avatar` component displays a user profile picture, initials, or fallback icon.

## Usage

```tsx
import { Avatar } from "@/components/ui/avatar";

export function ExampleAvatar() {
  return (
    <Avatar.Root size="md" shape="circle">
      <Avatar.Image src="https://images.unsplash.com/photo-1543610892-0b1f7e6d8ac1?w=128&h=128&dpr=2&q=80" />
      <Avatar.Fallback delay={600}>LT</Avatar.Fallback>
    </Avatar.Root>
  );
}
```

## API Reference

### Avatar.Root Props

| Prop        | Type                                | Default     | Description             |
| :---------- | :---------------------------------- | :---------- | :---------------------- |
| `size`      | `'sm' \| 'md' \| 'lg' \| 'xl'`      | `'md'`      | Avatar dimensions.      |
| `shape`     | `'circle' \| 'square' \| 'rounded'` | `'circle'`  | Border radius shape.    |
| `className` | `string`                            | `undefined` | Additional CSS classes. |

### Avatar.Fallback Props

| Prop    | Type     | Default     | Description                            |
| :------ | :------- | :---------- | :------------------------------------- |
| `delay` | `number` | `undefined` | Delay in ms before rendering fallback. |
