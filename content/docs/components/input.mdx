import { InputDemo } from "@/components/ui/input/input-demo";

The `Input` component renders an accessible form text input with customizable size variants and focus state styling.

## Usage

```tsx
import { Input } from "@/components/ui/input";

export function ExampleInput() {
  return (
    <label className="flex flex-col items-start gap-1 text-sm font-medium">
      Name
      <Input placeholder="e.g. Colm Tuite" />
    </label>
  );
}
```

## API Reference

### Input Props

| Prop            | Type                      | Default     | Description                        |
| :-------------- | :------------------------ | :---------- | :--------------------------------- |
| `size`          | `'sm' \| 'md' \| 'lg'`    | `'md'`      | Dimension size of the input.       |
| `value`         | `string`                  | `undefined` | Controlled input value.            |
| `defaultValue`  | `string`                  | `undefined` | Initial uncontrolled input value.  |
| `onValueChange` | `(value: string) => void` | `undefined` | Callback fired when value changes. |
| `disabled`      | `boolean`                 | `false`     | Disables user interaction.         |
