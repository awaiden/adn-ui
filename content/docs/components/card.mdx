import {
  CardRoot,
  CardHeader,
  CardTitle,
  CardDescription,
  CardContent,
  CardFooter,
} from "@/components/ui/card";
import { Button } from "@/components/ui/button";

The `Card` component displays content in a boxed container with support for compound slot subcomponents.

## Usage

```tsx
import { Card } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

export function CardExample() {
  return (
    <Card.Root>
      <Card.Header>
        <Card.Title>Notifications</Card.Title>
        <Card.Description>You have 3 unread messages.</Card.Description>
      </Card.Header>
      <Card.Content>
        <p>Your subscription is active and will renew next month.</p>
      </Card.Content>
      <Card.Footer>
        <Button variant="primary">Mark all as read</Button>
      </Card.Footer>
    </Card.Root>
  );
}
```

## Variants

### Default

The default card features a subtle background border and shadow.

    </CardHeader>
    <CardContent>
      <p className="text-sm">Card content goes here.</p>
    </CardContent>
  </CardRoot>
</div>

```tsx
<Card.Root variant="default">

  </Card.Header>
```

## Structure & Slots

The Card component uses a compound slot pattern:

- **`Card.Root` / `CardRoot`**: Main wrapper container providing variant context.
- **`Card.Header` / `CardHeader`**: Container for header elements (`flex flex-col gap-1.5 p-6`).
- **`Card.Title` / `CardTitle`**: Heading element for the card (`<h3>`).
- **`Card.Description` / `CardDescription`**: Subtitle paragraph (`<p>`).
- **`Card.Content` / `CardContent`**: Main content section (`p-6 pt-0`).
- **`Card.Footer` / `CardFooter`**: Footer container for action buttons or metadata (`flex items-center p-6 pt-0`).

## API Reference

### Card.Root Props

| Prop        | Type                     | Default     | Description                         |
| :---------- | :----------------------- | :---------- | :---------------------------------- |
| `variant`   | `'default' \| 'outline'` | `'default'` | Visual variant of the card.         |
| `className` | `string`                 | `undefined` | Additional CSS classes.             |
| `children`  | `React.ReactNode`        | `undefined` | Slot subcomponents and child nodes. |
